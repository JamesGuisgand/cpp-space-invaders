cmake_minimum_required(VERSION 3.24)
project(cpp-space-invaders VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(core STATIC src/World.cpp src/ScoreManager.cpp)
target_include_directories(core PUBLIC src)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE core)
  foreach(suite collisions scores progression simulation)
    add_test(NAME ${suite} COMMAND core-tests ${suite})
  endforeach()
endif()
option(BUILD_GAME "Build the SFML game" ON)
if(BUILD_GAME)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF)
  set(SFML_BUILD_AUDIO OFF)
  set(SFML_BUILD_NETWORK OFF)
  FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git
                       GIT_TAG 3.0.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(SFML)
  add_executable(space-invaders src/main.cpp src/Game.cpp)
  target_link_libraries(space-invaders PRIVATE core SFML::Graphics)
  add_custom_command(TARGET space-invaders POST_BUILD COMMAND ${CMAKE_COMMAND}
    -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:space-invaders>/assets")
endif()
